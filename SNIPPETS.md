## 7. Platform Abstraction

```typescript
// lib/platform.ts
const isCapacitor = () => !!(window as any).Capacitor

export const platform = {
  isCapacitor,
  isWeb: () => !isCapacitor(),

  async vibrate() {
    if (isCapacitor()) {
      const { Haptics } = await import('@capacitor/haptics')
      await Haptics.vibrate()
    } else {
      navigator.vibrate?.(50)
    }
  },

  async share(text: string) {
    if (isCapacitor()) {
      const { Share } = await import('@capacitor/share')
      await Share.share({ text })
    } else {
      await navigator.share?.({ text })
    }
  }
}
```

### Local Entry Adapter Example

```typescript
// services/modules/energy/local/energyEntryAdapter.ts
import { getDB } from '@/lib/idb'
import type { StorageAdapter } from '@/services/storage/types'
import type { EnergyEntry, EnergyEntryRecord } from '@/types'

const NOW = () => new Date().toISOString()

export const localEnergyEntryAdapter: StorageAdapter<EnergyEntry> = {
  async save(entry) {
    const db = await getDB()
    const record: EnergyEntryRecord = {
      ...entry,
      userId: 'local',        // replaced with real userId from authStore
      createdAt: NOW(),
      updatedAt: NOW(),
    }
    await db.put('energyEntries', record)
  },

  async getAll() {
    const db = await getDB()
    const records = await db.getAll('energyEntries')
    return records.map(({ userId, createdAt, updatedAt, ...entry }) => entry)
  },

  async delete(id) {
    const db = await getDB()
    await db.delete('energyEntries', id)
  }
}
```

### Supabase Entry Adapter Example

```typescript
// services/modules/energy/supabase/energyEntryAdapter.ts
import { supabase } from '@/lib/supabase'
import type { StorageAdapter } from '@/services/storage/types'
import type { EnergyEntry } from '@/types'

export const supabaseEnergyEntryAdapter: StorageAdapter<EnergyEntry> = {
  async save(entry) {
    const { error } = await supabase!.from('energy_entries').upsert(entry)
    if (error) throw error
  },

  async getAll() {
    const { data, error } = await supabase!
      .from('energy_entries')
      .select('*')
      .order('date', { ascending: false })
    if (error) throw error
    return data
  },

  // Extended query — date range (not in base interface)
  async getByDateRange(from: string, to: string): Promise<EnergyEntry[]> {
    const { data, error } = await supabase!
      .from('energy_entries')
      .select('*')
      .gte('date', from)
      .lte('date', to)
      .order('date', { ascending: false })
    if (error) throw error
    return data
  },

  async delete(id) {
    const { error } = await supabase!.from('energy_entries').delete().eq('id', id)
    if (error) throw error
  }
}
```

// router/guards.tsx
import { Navigate } from 'react-router-dom'
import { useAuthStore } from '@/store/authStore'
import { ROUTES } from './routes'
import type { ModuleId } from '@/types'

const onboardingRoute: Record<ModuleId, string> = {
  energy: ROUTES.ENERGY_ONBOARDING,
  pain: ROUTES.PAIN_ONBOARDING,
  mood: '/mood/onboarding',
}

export function ModuleGuard({
  moduleId,
  children
}: {
  moduleId: ModuleId
  children: React.ReactNode
}) {
  const onboardedModules = useAuthStore((s) => s.onboardedModules)
  if (!onboardedModules.includes(moduleId)) {
    return <Navigate to={onboardingRoute[moduleId]} replace />
  }
  return <>{children}</>
}

## 14. Storage Hook Example

```typescript
// modules/energy/hooks/useEnergyEntries.ts
import { useState, useEffect, useCallback } from 'react'
import { localEnergyEntryAdapter } from '@/services/modules/energy/local/energyEntryAdapter'
import type { EnergyEntry } from '@/types'
import { v4 as uuid } from 'uuid'

export function useEnergyEntries() {
  const [entries, setEntries] = useState<EnergyEntry[]>([])
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState<Error | null>(null)

  const load = useCallback(async () => {
    try {
      setLoading(true)
      const data = await localEnergyEntryAdapter.getAll()
      setEntries(data)
    } catch (e) {
      setError(e as Error)
    } finally {
      setLoading(false)
    }
  }, [])

  useEffect(() => { load() }, [load])

  const save = useCallback(async (
    templateId: string,
    cost: number,
    date: string,
    opts?: { duration?: number; notes?: string }
  ) => {
    const entry: EnergyEntry = {
      id: uuid(),
      templateId,
      date,
      timestamp: new Date().toISOString(),
      cost,
      ...opts,
    }
    await localEnergyEntryAdapter.save(entry)
    setEntries(prev => [entry, ...prev])
  }, [])

  const remove = useCallback(async (id: string) => {
    await localEnergyEntryAdapter.delete(id)
    setEntries(prev => prev.filter(e => e.id !== id))
  }, [])

  return { entries, loading, error, save, remove, reload: load }
}
```

```typescript
// services/sync/syncService.ts — outline
import { getDB } from '@/lib/idb'
import { supabase } from '@/lib/supabase'

export const syncService = {
  // Full sync on login — pull all remote records into local cache
  async fullSync(userId: string): Promise<void> {
    const tables = ['energyEntries', 'energyTemplates', 'painEntries', 'painTemplates', 'categories'] as const
    for (const table of tables) {
      const { data } = await supabase!.from(table).select('*')
      if (!data) continue
      const db = await getDB()
      const tx = db.transaction(table, 'readwrite')
      await Promise.all([
        ...data.map(record => tx.store.put({ ...record, syncStatus: 'synced' })),
        tx.done
      ])
    }
  },

  // Flush pending local writes to Supabase
  async flushPending(): Promise<void> {
    const db = await getDB()
    const tables = ['energyEntries', 'energyTemplates'] as const
    for (const table of tables) {
      const all = await db.getAll(table)
      const pending = all.filter(r => r.syncStatus === 'pending')
      for (const record of pending) {
        try {
          await supabase!.from(table).upsert(record)
          await db.put(table, { ...record, syncStatus: 'synced', syncedAt: new Date().toISOString() })
        } catch {
          // leave as pending, retry next time
        }
      }
    }
  }
}

// Wire up in main.tsx or authStore
window.addEventListener('online', () => syncService.flushPending())
```

// hooks/useSyncStatus.ts
import { useState, useEffect } from 'react'
import { getDB } from '@/lib/idb'

export function useSyncStatus() {
  const [pendingCount, setPendingCount] = useState(0)
  const [lastSynced, setLastSynced] = useState<string | null>(null)

  useEffect(() => {
    async function check() {
      const db = await getDB()
      const entries = await db.getAll('energyEntries')
      setPendingCount(entries.filter(e => e.syncStatus === 'pending').length)
      const synced = entries
        .filter(e => e.syncedAt)
        .map(e => e.syncedAt!)
        .sort()
        .at(-1) ?? null
      setLastSynced(synced)
    }
    check()
  }, [])

  return { pendingCount, lastSynced, isOnline: navigator.onLine }
}
